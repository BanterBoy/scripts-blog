# Example: Send SilverRail Application Invitations with New-EntraGuestInvitationEntra

# Example 1: Invite a SilverRail user with full profile and group assignment
New-EntraGuestInvitationEntra `
    -InvitedUserEmailAddress "silverrail.user1@example.com" `
    -InviteRedirectUrl "https://www.adminbyrequest.com/saml" `
    -InvitedUserDisplayName "SilverRail User One" `
    -CustomMessageBody "Welcome to SilverRail! Please follow the onboarding steps." `
    -GroupId "SR-Group-01","SR-Group-02" `
    -GivenName "User" `
    -Surname "One" `
    -Department "Operations" `
    -JobTitle "Rail Operator" `
    -CompanyName "SilverRail" `
    -MobilePhone "+44 7700 123456" `
    -StreetAddress "1 Rail Street" `
    -City "London" `
    -State "Greater London" `
    -PostalCode "E1 1AA" `
    -Country "United Kingdom" `
    -OtherMails "user1.alt@silverrail.com","user1.billing@silverrail.com" `
    -PassThru
